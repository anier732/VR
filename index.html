using UnityEngine; /// /// Meta Quest 3S向け：女の子とイチャイチャするための統合システム /// GitHub公開用 / コピペで即動作可能 /// public class VRFlirtSystem : MonoBehaviour { [Header("--- 参照設定 (References) ---")] [Tooltip("プレイヤーの頭（Main Camera）を割り当ててください")] public Transform playerHead; [Tooltip("女の子のアニメーターを割り当ててください")] public Animator girlAnimator; [Tooltip("女の子の頭のボーンを割り当ててください（目線追従用）")] public Transform girlHeadBone; [Header("--- なでなで設定 (Petting) ---")] [Tooltip("VRコントローラーのオブジェクトに設定するタグ")] public string controllerTag = "GameController"; [Tooltip("なでる判定にするための、手の最低移動速度")] public float petVelocityThreshold = 0.1f; [Header("--- 距離感設定 (Proximity) ---")] [Tooltip("これ以上近づくと『限界接近』と判定する距離（メートル）")] public float closeDistance = 0.4f; // 内部計算用 private Vector3 lastHandPosition; private bool isPetting = false; private bool isTooClose = false; void Start() { // アニメーターが未設定なら自動取得を試みる if (girlAnimator == null) girlAnimator = GetComponent(); // プレイヤー頭部が未設定ならMainCameraを自動取得 if (playerHead == null && Camera.main != null) playerHead = Camera.main.transform; } void Update() { if (playerHead == null) return; // 1. 目線追従（ルックアット機能） if (girlHeadBone != null) { Vector3 targetPos = playerHead.position; // 首が不自然に傾かないように高さを固定 targetPos.y = girlHeadBone.position.y; girlHeadBone.LookAt(targetPos); } // 2. 距離判定（近づくとドキドキ） float distance = Vector3.Distance(transform.position, playerHead.position); if (distance < closeDistance && !isTooClose) { isTooClose = true; girlAnimator.SetBool("IsTooClose", true); Debug.Log("女の子：『ひゃぅ！？ ち、近いよぉ…///』"); } else if (distance >= closeDistance && isTooClose) { isTooClose = false; girlAnimator.SetBool("IsTooClose", false); } } // 3. なでなで判定（頭のColliderにコントローラーが触れている時） private void OnTriggerStay(Collider other) { if (other.CompareTag(controllerTag)) { // コントローラーの動く速度を計算 float handSpeed = (other.transform.position - lastHandPosition).magnitude / Time.deltaTime; lastHandPosition = other.transform.position; if (handSpeed > petVelocityThreshold && !isPetting) { isPetting = true; girlAnimator.SetBool("IsPetting", true); Debug.Log("女の子：『んぅ…あたま、なでなでするの、あったかくて好き……』"); } } } private void OnTriggerExit(Collider other) { if (other.CompareTag(controllerTag)) { isPetting = false; girlAnimator.SetBool("IsPetting", false); Debug.Log("女の子：『あ…止めちゃうの？ もっと、なでて…？』"); } } }
